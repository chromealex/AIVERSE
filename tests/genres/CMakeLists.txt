cmake_minimum_required(VERSION 3.22)
project(RtsGenreTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
add_library(genre_core STATIC ${ROOT}/engine/core/Fixed.cpp ${ROOT}/engine/core/FixedMath.cpp ${ROOT}/engine/data/Json.cpp ${ROOT}/engine/sim/Simulation.cpp)
target_include_directories(genre_core PUBLIC ${ROOT})
add_library(genre_board STATIC ${ROOT}/modules/genres/BoardModule.cpp)
target_link_libraries(genre_board PUBLIC genre_core)
add_library(genre_motion STATIC ${ROOT}/modules/genres/MotionModule.cpp ${ROOT}/modules/fps/FpsMotor.cpp ${ROOT}/engine/world/SpatialWorld.cpp ${ROOT}/engine/world/TriangleCollision.cpp)
target_link_libraries(genre_motion PUBLIC genre_core)
add_executable(aiverse_genre_tests tests.cpp)
target_link_libraries(aiverse_genre_tests PRIVATE genre_board genre_motion)
if(NOT MSVC)
    target_compile_options(genre_core PRIVATE -Wall -Wextra -Werror)
    target_compile_options(genre_board PRIVATE -Wall -Wextra -Werror)
    target_compile_options(genre_motion PRIVATE -Wall -Wextra -Werror)
    target_compile_options(aiverse_genre_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME genres COMMAND aiverse_genre_tests ${ROOT}/examples/genres)
