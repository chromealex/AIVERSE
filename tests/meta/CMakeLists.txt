cmake_minimum_required(VERSION 3.22)
project(AiverseMetaTests LANGUAGES CXX)
enable_testing()
add_subdirectory(../../engine/meta meta)
add_executable(aiverse_meta_tests tests.cpp)
target_link_libraries(aiverse_meta_tests PRIVATE aiverse::meta)
if(NOT MSVC)
    target_compile_options(aiverse_meta PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(aiverse_meta_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME aiverse_meta_tests COMMAND aiverse_meta_tests)
add_executable(aiverse_meta_validate ../../tools/meta/validate.cpp)
target_link_libraries(aiverse_meta_validate PRIVATE aiverse::meta)
foreach(pair economy:meta boards:boards chats:chats)
    string(REPLACE ":" ";" parts "${pair}")
    list(GET parts 0 kind)
    list(GET parts 1 filename)
    add_test(NAME preset_${kind} COMMAND aiverse_meta_validate ${kind} ${CMAKE_CURRENT_LIST_DIR}/../../examples/meta/${filename}.json)
endforeach()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME meta_native_link COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/native_link.py ${CMAKE_COMMAND})
endif()
if(Python3_Interpreter_FOUND)
    add_test(NAME meta_scaffold COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/scaffold.py $<TARGET_FILE:aiverse_meta_validate>)
endif()

add_executable(aiverse_meta_service ../../tools/meta/service_main.cpp)
target_link_libraries(aiverse_meta_service PRIVATE aiverse::meta)
add_executable(aiverse_meta_service_tests service_tests.cpp)
target_link_libraries(aiverse_meta_service_tests PRIVATE aiverse::meta)
if(NOT MSVC)
    target_compile_options(aiverse_meta_service_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME meta_service COMMAND aiverse_meta_service_tests ${CMAKE_CURRENT_LIST_DIR}/../../examples/meta/service.json)
add_test(NAME preset_service COMMAND aiverse_meta_validate service ${CMAKE_CURRENT_LIST_DIR}/../../examples/meta/service.json)
if(Python3_Interpreter_FOUND)
    add_test(NAME meta_http_backend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/backend_test.py $<TARGET_FILE:aiverse_meta_service>)
endif()

add_executable(aiverse_meta_liveops_tests liveops_tests.cpp)
target_link_libraries(aiverse_meta_liveops_tests PRIVATE aiverse::meta)
if(NOT MSVC)
    target_compile_options(aiverse_meta_liveops_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME meta_liveops COMMAND aiverse_meta_liveops_tests ${CMAKE_CURRENT_LIST_DIR}/../../examples/meta/service.json ${CMAKE_CURRENT_LIST_DIR}/../../examples/meta/liveops.json)
if(Python3_Interpreter_FOUND)
    add_test(NAME meta_liveops_schema COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../../tools/meta/generate_liveops.py --check)
endif()

add_executable(aiverse_meta_client_tests client_tests.cpp)
target_link_libraries(aiverse_meta_client_tests PRIVATE aiverse::meta)
if(NOT MSVC)
    target_compile_options(aiverse_meta_client_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME meta_client COMMAND aiverse_meta_client_tests)
if(Python3_Interpreter_FOUND)
    add_test(NAME meta_kits COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/kits.py $<TARGET_FILE:aiverse_meta_validate>)
endif()

if(Python3_Interpreter_FOUND)
    add_test(NAME meta_database COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/database_test.py)
endif()

if(Python3_Interpreter_FOUND)
    add_test(NAME skill_catalog COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../../tools/skills/catalog.py check)
    add_test(NAME skill_routing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../skills/catalog_test.py)
    if(UNIX AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME skill_library_isolation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../skills/isolation.py)
    set_tests_properties(skill_library_isolation PROPERTIES TIMEOUT 600)
    endif()
