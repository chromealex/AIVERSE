cmake_minimum_required(VERSION 3.16)
project(VividDnaImport LANGUAGES CXX)
set(DNA_SDK_ROOT "" CACHE PATH "External Epic MetaHuman-DNA-Calibration checkout")
if(NOT EXISTS "${DNA_SDK_ROOT}/dnacalib/DNACalib/include/dna/BinaryStreamReader.h")
  message(FATAL_ERROR "Set DNA_SDK_ROOT to your external DNA Calibration checkout")
endif()
list(APPEND CMAKE_MODULE_PATH "${DNA_SDK_ROOT}/dnacalib/CMakeModulesExtra")
set(DNAC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${DNA_SDK_ROOT}/dnacalib/DNACalib" sdk)
add_executable(vivid_dna_decode decode.cpp)
target_compile_features(vivid_dna_decode PRIVATE cxx_std_17)
target_link_libraries(vivid_dna_decode PRIVATE DNACalib::dnacalib)
