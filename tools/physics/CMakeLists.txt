cmake_minimum_required(VERSION 3.24)
project(AIVERSEPhysicsEvaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# This standalone target never adds Jolt to the engine or downloads it implicitly.
set(AIVERSE_JOLT_ARCHIVE "" CACHE FILEPATH "Pinned Jolt source archive")
if(NOT EXISTS "${AIVERSE_JOLT_ARCHIVE}")
  message(FATAL_ERROR "Provide -DAIVERSE_JOLT_ARCHIVE=/path/to/jolt.tar.gz (see README.md)")
endif()
file(SHA256 "${AIVERSE_JOLT_ARCHIVE}" archive_hash)
if(NOT archive_hash STREQUAL "834644d7e9dffee9be0220748015667649d3361611b01c7e6839a5eb58e89711")
  message(FATAL_ERROR "Jolt archive SHA256 mismatch")
endif()
include(FetchContent)
foreach(flag GENERATE_DEBUG_SYMBOLS INTERPROCEDURAL_OPTIMIZATION DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE ENABLE_INSTALL ENABLE_ALL_WARNINGS)
  set(${flag} OFF CACHE BOOL "Evaluation configuration" FORCE)
endforeach()
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "Evaluation configuration" FORCE)
FetchContent_Declare(jolt URL "${AIVERSE_JOLT_ARCHIVE}" URL_HASH SHA256=834644d7e9dffee9be0220748015667649d3361611b01c7e6839a5eb58e89711 SOURCE_SUBDIR Build DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(jolt)
add_executable(aiverse_physics_bench main.cpp)
target_link_libraries(aiverse_physics_bench PRIVATE Jolt)
if(MSVC)
  target_compile_options(aiverse_physics_bench PRIVATE /W4 /fp:precise)
else()
  target_compile_options(aiverse_physics_bench PRIVATE -Wall -Wextra -ffp-contract=off)
endif()
enable_testing()
add_test(NAME physics_collision COMMAND aiverse_physics_bench verify 1 0)
