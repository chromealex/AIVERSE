# All implementation archives are opt-in. Merely adding this directory builds none.
function(aiverse_meta_component name)
    add_library(aiverse_meta_${name} STATIC EXCLUDE_FROM_ALL ${ARGN})
    target_link_libraries(aiverse_meta_${name} PUBLIC aiverse_meta_data)
    set_target_properties(aiverse_meta_${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(AIVERSE::meta_${name} ALIAS aiverse_meta_${name})
    if(NOT MSVC)
        target_compile_options(aiverse_meta_${name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

# Generate a typed factory whose references are the complete selection. No runtime discovery.
function(aiverse_add_meta_server target)
    cmake_parse_arguments(KIT "" "" "MODULES" ${ARGN})
    if(KIT_UNPARSED_ARGUMENTS OR NOT KIT_MODULES OR NOT target MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
        message(FATAL_ERROR "aiverse_add_meta_server(target MODULES economy ...) requires a valid target and explicit modules")
    endif()
    file(READ "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/modules.json" manifest)
    string(JSON count LENGTH "${manifest}" modules)
    math(EXPR last "${count}-1")
    set(known "")
    foreach(i RANGE ${last})
        string(JSON id GET "${manifest}" modules ${i} id)
        list(APPEND known "${id}")
        string(JSON register GET "${manifest}" modules ${i} register)
        set(register_${id} "${register}")
        string(JSON count_deps LENGTH "${manifest}" modules ${i} dependencies)
        set(deps_${id} "")
        if(count_deps GREATER 0)
            math(EXPR last_dep "${count_deps}-1")
            foreach(j RANGE ${last_dep})
                string(JSON dep GET "${manifest}" modules ${i} dependencies ${j})
                list(APPEND deps_${id} "${dep}")
            endforeach()
        endif()
    endforeach()
    set(selected economy ${KIT_MODULES})
    set(pending ${selected})
    while(pending)
        list(POP_FRONT pending part)
        if(NOT part IN_LIST known)
            message(FATAL_ERROR "Unknown meta policy: ${part}")
        endif()
        foreach(dep IN LISTS deps_${part})
            if(NOT dep IN_LIST selected)
                list(APPEND selected ${dep})
                list(APPEND pending ${dep})
            endif()
        endforeach()
    endwhile()
    list(REMOVE_DUPLICATES selected)
    set(code "#include <engine/meta/Modules.h>\nnamespace aiverse::meta::kits { Modules ${target}() { Modules m;\n")
    foreach(part IN LISTS selected)
        set(function_name "${register_${part}}")
        string(APPEND code "modules::${function_name}(m);\n")
    endforeach()
    string(APPEND code "return m; } }\n")
    set(dir "${CMAKE_CURRENT_BINARY_DIR}/${target}_generated")
    file(MAKE_DIRECTORY "${dir}")
    file(WRITE "${dir}/${target}.cpp" "${code}")
    file(WRITE "${dir}/${target}.h" "#pragma once\n#include <engine/meta/Modules.h>\nnamespace aiverse::meta::kits { Modules ${target}(); }\n")
    add_library(${target} STATIC EXCLUDE_FROM_ALL "${dir}/${target}.cpp")
    target_include_directories(${target} PUBLIC "${dir}")
    target_link_libraries(${target} PUBLIC AIVERSE::meta_server)
    foreach(part IN LISTS selected)
        string(REPLACE "-" "_" name "${part}")
        target_link_libraries(${target} PUBLIC aiverse_meta_${name})
    endforeach()
endfunction()

# Native OBJECT modules are inserted with TARGET_OBJECTS: propagate dependencies to owner.
function(aiverse_native_use_meta native_target)
    if(NOT TARGET ${native_target} OR NOT TARGET aiverse_game)
        message(FATAL_ERROR "aiverse_native_use_meta expects a registered native target")
    endif()
    set(side "${ARGV1}")
    if(side STREQUAL "CLIENT")
        set(dependency AIVERSE::meta_client)
    elseif(side STREQUAL "SERVER")
        set(dependency AIVERSE::meta_server)
    elseif(TARGET "${side}")
        set(dependency "${side}")
    elseif(side STREQUAL "")
        set(dependency aiverse::meta)
    else()
        message(FATAL_ERROR "meta selection must be CLIENT, SERVER or an existing selected kit target")
    endif()
    target_link_libraries(${native_target} PRIVATE ${dependency})
    target_link_libraries(aiverse_game PUBLIC ${dependency})
endfunction()
