# OBJECT native modules are inserted with TARGET_OBJECTS: their PRIVATE link requirements do not
# propagate. Put backend libraries on the owning game target, avoiding a native -> aiverse_game cycle.
function(aiverse_native_use_bots target)
    if(NOT TARGET ${target} OR NOT TARGET aiverse_game)
        message(FATAL_ERROR "aiverse_native_use_bots requires a registered native target")
    endif()
    target_link_libraries(${target} PRIVATE AIVERSE::gameplay_native)
    foreach(genre IN LISTS ARGN)
        if(genre STREQUAL "FPS")
            target_link_libraries(aiverse_game PUBLIC aiverse_fps)
        elseif(genre STREQUAL "NAVMESH")
            target_link_libraries(aiverse_game PUBLIC aiverse_world)
        elseif(NOT genre STREQUAL "RTS")
            message(FATAL_ERROR "aiverse_native_use_bots: expected RTS, FPS or NAVMESH")
        endif()
    endforeach()
endfunction()

# Per-skill packages. INTERFACE targets emit no objects and have only explicit dependencies.
file(READ "${CMAKE_CURRENT_LIST_DIR}/catalog.json" _aiverse_catalog)
string(JSON _aiverse_count LENGTH "${_aiverse_catalog}" packages)
math(EXPR _aiverse_last "${_aiverse_count}-1")
foreach(i RANGE ${_aiverse_last})
    string(JSON runtime GET "${_aiverse_catalog}" packages ${i} runtime)
    if(NOT runtime STREQUAL "header")
        continue()
    endif()
    string(JSON id GET "${_aiverse_catalog}" packages ${i} id)
    string(REPLACE "-" "_" name "${id}")
    add_library(aiverse_lib_${name} INTERFACE)
    add_library(AIVERSE::lib_${name} ALIAS aiverse_lib_${name})
    target_compile_features(aiverse_lib_${name} INTERFACE cxx_std_20)
    target_include_directories(aiverse_lib_${name} INTERFACE "${CMAKE_CURRENT_LIST_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}/..")
    foreach(field linkDependencies packageDependencies)
        string(JSON count LENGTH "${_aiverse_catalog}" packages ${i} ${field})
        if(count GREATER 0)
            math(EXPR last "${count}-1")
            foreach(j RANGE ${last})
                string(JSON dep GET "${_aiverse_catalog}" packages ${i} ${field} ${j})
                if(field STREQUAL "packageDependencies")
                    string(REPLACE "-" "_" dep "${dep}")
                    set(dep AIVERSE::lib_${dep})
                endif()
                target_link_libraries(aiverse_lib_${name} INTERFACE ${dep})
            endforeach()
        endif()
    endforeach()
endforeach()
function(aiverse_native_use_libraries target)
    if(NOT TARGET ${target} OR NOT TARGET aiverse_game OR NOT ARGN)
        message(FATAL_ERROR "aiverse_native_use_libraries expects a native target and package IDs")
    endif()
    foreach(id IN LISTS ARGN)
        string(REPLACE "-" "_" name "${id}")
        if(NOT TARGET AIVERSE::lib_${name})
            message(FATAL_ERROR "Unknown code library: ${id}")
        endif()
        target_link_libraries(${target} PRIVATE AIVERSE::lib_${name})
        target_link_libraries(aiverse_game PUBLIC AIVERSE::lib_${name})
    endforeach()
endfunction()
