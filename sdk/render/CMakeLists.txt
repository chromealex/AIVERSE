add_library(aiverse_render SHARED aiverse_render.cpp)
target_compile_definitions(aiverse_render PRIVATE AIVERSE_RENDER_EXPORTS)
target_include_directories(aiverse_render PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(aiverse_render PRIVATE aiverse_headless_model_frame)
set_target_properties(aiverse_render PROPERTIES VERSION 1.0 SOVERSION 1)
find_package(JNI QUIET)
if(JNI_FOUND)
    add_library(aiverse_render_jni SHARED java/Jni.cpp)
    target_include_directories(aiverse_render_jni PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(aiverse_render_jni PRIVATE aiverse_render)
endif()

include(GNUInstallDirs)
install(TARGETS aiverse_render LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(TARGET aiverse_render_jni)
    if(APPLE)
        set_target_properties(aiverse_render_jni PROPERTIES INSTALL_RPATH "@loader_path")
    elseif(UNIX)
        set_target_properties(aiverse_render_jni PROPERTIES INSTALL_RPATH "$ORIGIN")
    endif()
    install(TARGETS aiverse_render_jni LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES aiverse_render.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/aiverse)
install(FILES python/aiverse_render.py DESTINATION ${CMAKE_INSTALL_DATADIR}/aiverse/render/python)
install(FILES csharp/RenderModel.cs DESTINATION ${CMAKE_INSTALL_DATADIR}/aiverse/render/csharp)
install(FILES java/org/aiverse/render/RenderModel.java DESTINATION ${CMAKE_INSTALL_DATADIR}/aiverse/render/java/org/aiverse/render)
